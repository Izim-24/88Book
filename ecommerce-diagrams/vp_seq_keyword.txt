actor Customer
Customer->SearchUI: selectSearchBar()
SearchUI->SearchHistoryRepository: getRecentSearches(customerId)
SearchHistoryRepository-->SearchUI: recentKeywords
SearchUI->ProductRepository: getTrendingKeywords()
ProductRepository-->SearchUI: trendingKeywords
SearchUI-->Customer: showSuggestedKeywords(recent + trending)
Customer->SearchUI: enterKeyword(keyword)
SearchUI->SearchUI: validateFormat(keyword)

alt invalid format
    SearchUI-->Customer: showInvalidKeywordMessage("Keyword must be 2-50 characters")
else valid format
    Customer->SearchUI: submitKeyword(keyword)
    SearchUI->SearchController: searchByKeyword(keyword)
    SearchController->SearchService: findByKeyword(keyword)
    SearchService->SearchService: normalizeKeyword(keyword)
    SearchService->ProductRepository: findByKeyword(normalizedKeyword)
    ProductRepository-->SearchService: productList

    alt no products found
        SearchService-->SearchController: emptyResult
        SearchController-->SearchUI: showNoProductsFound()
        SearchUI-->Customer: display "No products found"
    else products found
        SearchService->SearchHistoryRepository: saveSearchHistory(customerId, keyword)
        SearchHistoryRepository-->SearchService: saved
        SearchService-->SearchController: searchResult
        SearchController-->SearchUI: displayProducts(productList)
        SearchUI-->Customer: show product list
    end
end
