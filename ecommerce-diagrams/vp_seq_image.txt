    ImageSearchUI->PermissionManager: requestPermission()
    PermissionManager-->Customer: showPermissionDialog()
    Customer-->PermissionManager: grantPermission()
    PermissionManager-->ImageSearchUI: granted

    ImageSearchUI-->Customer: showFileError("Only JPG/PNG, max 10MB")
else valid file
    ImageSearchUI->ImageAnalysisService: extractFeatures(image)
    ImageAnalysisService->ImageAnalysisService: preprocessImage()
    ImageAnalysisService->ImageAnalysisService: detectObjects()
    ImageAnalysisService-->ImageSearchUI: visualFeatures
    ImageSearchUI-->Customer: previewImage(image)
    Customer->ImageSearchUI: markTargetObject()
    Customer->ImageSearchUI: confirmSelection()
    ImageSearchUI->SearchController: searchSimilar(features)
    SearchController->SearchService: findSimilarProducts(features)
    SearchService->ProductRepository: matchByFeatures(features)
    ProductRepository-->SearchService: productList

    alt no similar product found
        SearchService-->SearchController: emptyResult
        SearchController-->ImageSearchUI: showNoMatchMessage()
        ImageSearchUI-->Customer: display "No matching products"
    else products found
        SearchService->SearchHistoryRepository: saveSearchHistory(customerId, image, features)
        SearchHistoryRepository-->SearchService: saved
        SearchService-->SearchController: searchResult
        SearchController-->ImageSearchUI: displayResultList(productList)
        ImageSearchUI-->Customer: show result list
    end
